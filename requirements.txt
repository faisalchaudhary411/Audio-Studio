# Studio-quality denoise (DeepFilterNet AI speech enhancement).
# TWO one-time VPS setup steps are REQUIRED before `pip install -r
# requirements.txt` will succeed here -- do both, in order, every time
# this is set up on a fresh VPS/venv:
#
# 1. Install Rust (deepfilternet's DeepFilterLib component has no
#    pre-built wheel for most Linux/Python combos -- pip has to compile
#    it from source, which needs cargo on PATH):
#      curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh -s -- -y
#      source "$HOME/.cargo/env"
#
# 2. Install the pinned CPU-only torch + torchaudio BEFORE running this
#    file. This VPS has no GPU; plain `pip install torch` / `torchaudio`
#    from PyPI pulls multi-GB CUDA builds. The versions below are pinned
#    because newer torchaudio (2.9+) removed APIs that the published
#    deepfilternet package still imports (torchaudio.backend / AudioMetaData).
#    Verified working pair (2026-09):
#      pip install torch==2.5.1 torchaudio==2.5.1 --index-url https://download.pytorch.org/whl/cpu
#
# Skip step 1 and deepfilternet's build fails with "Cargo ... is not
# installed". Skip step 2 and a plain `pip install -r requirements.txt`
# silently drags in ~5GB of unused CUDA packages or a too-new torchaudio
# that breaks Studio denoise.
torch==2.5.1
torchaudio==2.5.1
deepfilternet>=0.5.6
